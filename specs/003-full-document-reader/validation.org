// Native-test fixture recipe. Synthetic text only; never reads the user's vault.
import fs from "node:fs";
import path from "node:path";
import { tmpdir } from "node:os";
import { createHash } from "node:crypto";
const destination = fs.mkdtempSync(path.join(tmpdir(), "full-document-reader-fixtures-"));
const sentence = "The morning tide carried small boats past the harbour while people watched from the stone wall in quiet sunlight.".split(" ");
const words = Array.from({ length: 100000 }, (_, index) => sentence[index % sentence.length]);
words[0] = "BEGIN-READER-003";
words[50000] = "MIDDLE-READER-003";
words[99999] = "END-READER-003";
const paragraphs = [];
for (let index = 0; index < words.length; index += 100) paragraphs.push(words.slice(index, index + 100).join(" "));
const fixtures = {
  "Empty.md": "",
  "Second.md": "# A different note\n\nSecond-note text ends here.\n",
  "Mixed.md": "---\ntitle: Mixed reader fixture\n---\n\n# Beginning\n\nThis is a **short** complete reading sample, with a [local link](Second.md).\n\n- A list item\n- [ ] A task that must stay unchanged\n\n> A quoted sentence.\n\n| Place | Item |\n| --- | --- |\n| Harbour | Boat |\n\n\`\`\`text\nA code sample.\n\`\`\`\n\nA footnote reference.[^1]\n\n![[Second]]\n\n[^1]: The local footnote.\n\nThe mixed document ends with this sentence.\n",
  "Long.md": paragraphs.join("\n\n") + "\n"
};
const files = {};
for (const [name, text] of Object.entries(fixtures)) {
  fs.writeFileSync(path.join(destination, name), text);
  files[name] = { words: text.match(/\S+/gu)?.length ?? 0, sha256: createHash("sha256").update(text).digest("hex") };
}
console.log(JSON.stringify({ destination, files }, null, 2));
